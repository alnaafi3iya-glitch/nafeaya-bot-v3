# nafeaya-bot-v3 (خدمة تجريبية — لا تُرفع إلى nafeaya-bot-v2)

الملفات: bot.py + roles_module.py + requirements.txt + Procfile

## متغيرات Railway
BOT_TOKEN
PANEL_URL=رابط اللوحة (بدون / في النهاية)
PANEL_API_KEY
TEST_MODE=true
ROLES_ENABLED=true
ROLES_ADMIN_IDS=معرّفك الرقمي (وللمشرفات بفواصل)
ROLES_ALLOWED_CHAT_IDS=-1004470982964
ROLES_DB_PATH=/data/roles.db
REMINDER_STATE_DB=/data/state.sqlite3

أضيفي Volume على /data.

ROLES_TELEGRAM_ADMINS=true   (اختياري؛ الافتراضي true: مشرفات المجموعة في تيليغرام = مديرات الجلسات)

## اكتشاف المجموعات (جديد)
GROUPS_DISCOVERY=true   (اختياري؛ الافتراضي true ما دام PANEL_URL وPANEL_API_KEY مضبوطين)
- حين يُضاف البوت إلى مجموعة تظهر في لوحة الإدارة (صفحة «المجموعات») بانتظار الاعتماد، ولا يُرسل إليها شيء قبله.
- المجموعات المعرّفة في الكود (GROUPS) تُنقل إلى اللوحة معتمدة تلقائيًا عند أول قراءة للحصص.
- للمجموعات التي أُضيف إليها البوت قبل هذه الميزة: اكتبي /register_group داخل المجموعة (للمعرّفات في ROLES_ADMIN_IDS فقط).
- يعمل الاكتشاف بالاستقصاء (polling) في خدمة البوت v3 نفسها؛ تأكدي ألا تعمل خدمتان بالتوكن نفسه بالاستقصاء معًا.

## مواضيع المجموعات (Topics) — جديد
TOPICS_DISCOVERY=true   (اختياري؛ الافتراضي true مع اكتشاف المجموعات)
- في مجموعة فيها مواضيع (مثل تسميع حفص) يرسل البوت التذكير إلى موضوع المعلمة لا إلى المحادثة العامة.
- يكتشف البوت الموضوع حين تُكتب فيه أي رسالة (يجب أن يكون البوت مشرفًا ليرى الرسائل)، فيظهر في «المجموعات» تحت المجموعة المعتمدة بانتظار الاعتماد.
- بعد الاعتماد باسم (مثل أم_سلمة) يُكتب الاسم في حقل «المجموعات» للحصة كأي مجموعة.
- لا يُسجَّل موضوع في مجموعة غير معتمدة.

## اختبار الإرسال إلى موضوع
داخل المجموعة التي فيها الموضوع، ومن حساب في ROLES_ADMIN_IDS، وفي وضع TEST_MODE=true فقط:
/test_dest اسم_الموضوع
يرسل رسالة تجربة إلى ذلك الموضوع بمعرّفه؛ لا يرسل إلا إلى وجهة داخل المجموعة التي كُتب فيها الأمر.
